\section{Introducción}
% =========================================================

\begin{frame}{Sistema}
    \begin{columns}[c,onlytextwidth]
        \begin{column}{0.50\textwidth}
            \large
            \begin{itemize}\setlength{\itemsep}{1.0em}
                \item Billar \textbf{circular} de radio $R$ con pared fija y dos
                      \textbf{obstáculos} fijos de radio $r$ en $(\pm x_o, 0)$.
                \item $N$ partículas de radio $r$ y masa $m$, rapidez inicial $v_0$
                      y dirección al azar.
                \item Interactúan \textbf{sólo por contacto}, sin fricción ni
                      disipación.
                \item Nacen \textcolor{fresca}{\textbf{frescas}} y pasan a
                      \textcolor{usada}{\textbf{usadas}} al tocar \textbf{por primera
                      vez} un obstáculo; siguen moviéndose igual, $N$ no cambia.
            \end{itemize}
        \end{column}\hfill
        \begin{column}{0.47\textwidth}
            \centering
            \input{figuras/sistema.tikz}\\[0.4em]
            {\footnotesize\color{gris}
             \textcolor{fresca}{$\bullet$} fresca \quad
             \textcolor{usada}{$\bullet$} usada \quad
             \textcolor{obstaculo}{$\bullet$} obstáculo}
        \end{column}
    \end{columns}
\end{frame}


\begin{frame}{Modelo: contacto elástico y pared como partícula imagen}
    \begin{columns}[c,onlytextwidth]
        \begin{column}{0.58\textwidth}
            \small
            \setlength{\abovedisplayskip}{4pt}\setlength{\belowdisplayskip}{4pt}
            Newton para cada partícula, $m\,\ddot{\vr}_i = \vF_i$, con un
            \textbf{resorte lineal} normal, activo sólo mientras hay solapamiento:
            \[
                \vF_i = \sum_j -k\,\xi_{ij}\,\ehat_{ij}, \qquad
                \xi_{ij} = r_i + r_j - |\vr_j - \vr_i| > 0, \qquad
                \ehat_{ij} = \frac{\vr_j - \vr_i}{|\vr_j - \vr_i|}
            \]
            Los $j$ son las otras partículas, los dos obstáculos (fijos) y la
            partícula imagen de la pared.
            \vspace{0.5em}

            \rotulo{Pared.} Hay contacto si $|\vr_i| > R - r$; la normal es radial,
            $\nhat = \vr_i/|\vr_i|$, y el choque es contra una partícula virtual fija
            de radio $r$ que toca la pared en $C = R\,\nhat$:
            \[
                \vr_{\text{img}} = (R + r)\,\nhat, \qquad
                \xi_{iw} = |\vr_i| + r - R .
            \]
            La fuerza queda puramente normal; $\vr_{\text{img}}$ se recalcula en cada
            paso (la partícula se desplaza tangencialmente durante el choque).
        \end{column}\hfill
        \begin{column}{0.40\textwidth}
            \centering
            \input{figuras/modelo-pared-imagen.tikz}\\[0.3em]
            {\footnotesize\color{gris} esquema no a escala}
        \end{column}
    \end{columns}
\end{frame}

\section{Implementación}
% =========================================================

\begin{frame}{Integración de las ecuaciones de movimiento}
    \vspace{-0.3em}
    \begin{columns}[t,onlytextwidth]
        \begin{column}{0.44\textwidth}
            \rotulo{Billar (sistema 2): Verlet original} \cita{lo pide el enunciado}
            \vspace{-0.4em}
            \[\begin{aligned}
                \vr(t{+}\dt) &= 2\vr(t) - \vr(t{-}\dt) + \tfrac{\dt^2}{m}\vF(t)\\[0.2em]
                \vv(t) &= \frac{\vr(t{+}\dt) - \vr(t{-}\dt)}{2\dt}
            \end{aligned}\]
            \vspace{-0.6em}
            {\small
            \begin{itemize}\setlength{\itemsep}{0.25em}
                \item Paso fijo $\dt$; el estado se guarda cada $\dtd = n\,\dt$,
                      con $n$ entero.
                \item Arranque: $\vr(-\dt)$ con Euler en $-\dt$. $\vF$ depende sólo
                      de las posiciones: paso explícito.
                \item Tres posiciones por partícula: $\vr(t)$ y $\vv(t)$ del
                      \textbf{mismo instante}.
                \item \rotulo{Decisión} (oscilador): $\vv(t{+}\dt)$ \textbf{despejada}
                      ($\vF$ lineal en $\vv$), $O(\dt^2)$.
            \end{itemize}}
        \end{column}\hfill
        \begin{column}{0.53\textwidth}
            \rotulo{Oscilador (sistema 1): además}
            \cita{$\va(t{+}\dt)$ necesita $\vv(t{+}\dt)$: hay que elegir con cuál}
            \vspace{-0.5em}
            {\scriptsize
            \[\begin{aligned}
                &\textbf{Velocity Verlet}\\
                &\vr(t{+}\dt) = \vr(t) + \dt\,\vv(t) + \tfrac{\dt^2}{2}\va(t), \quad
                 \vv(t{+}\tfrac{\dt}{2}) = \vv(t) + \tfrac{\dt}{2}\va(t)\\
                &\vv(t{+}\dt) = \vv(t{+}\tfrac{\dt}{2}) + \tfrac{\dt}{2}\va(t{+}\dt)\\
                &\decision{la de \textbf{medio paso} $\vv(t{+}\tfrac{\dt}{2})$, $O(\dt)$:
                            \alert{pierde un orden}}\\[0.15em]
                &\textbf{Beeman (predictor-corrector)}\\
                &\vr(t{+}\dt) = \vr(t) + \vv(t)\dt + \tfrac{2}{3}\va(t)\dt^2 - \tfrac{1}{6}\va(t{-}\dt)\dt^2\\
                &\vv^{\,p}(t{+}\dt) = \vv(t) + \tfrac{3}{2}\va(t)\dt - \tfrac{1}{2}\va(t{-}\dt)\dt\\
                &\vv(t{+}\dt) = \vv(t) + \tfrac{1}{3}\va(t{+}\dt)\dt + \tfrac{5}{6}\va(t)\dt - \tfrac{1}{6}\va(t{-}\dt)\dt\\
                &\decision{la \textbf{predicha} $\vv^{\,p}(t{+}\dt)$, $O(\dt^2)$}\\[0.15em]
                &\textbf{Euler predictor-corrector}\\
                &\vv^{\,p} = \vv(t) + \va(t)\dt, \quad \vr^{\,p} = \vr(t) + \vv(t)\dt\\
                &\vv(t{+}\dt) = \vv(t) + \va(t{+}\dt)\dt, \quad \vr(t{+}\dt) = \vr(t) + \vv(t{+}\dt)\dt\\
                &\decision{la \textbf{de Euler} $\vv^{\,p}$, $O(\dt)$}
            \end{aligned}\]}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Observables}
    \vspace{-0.3em}
    \footnotesize
    \begin{itemize}\setlength{\itemsep}{0.45em}
        \item \rotulo{Conversión}: $\Fu(t) = \Nu(t)/N$, con $\Nu$ las usadas
              acumuladas. \alert{$t_x$} es el instante en que la conversión llega
              al $x$ \%, el que cumple $\Fu(t_x) = x/100$, tomado en el instante
              exacto de esa conversión; medimos $\tnov$ y $\tcien$.
        \item \rotulo{Energía total}: $E(t) = \sum_{i} \tfrac{1}{2} m v_i^2
              + \sum_{\text{contactos}} \tfrac{1}{2} k\,\xi^2$, constante si el
              esquema integra bien. Se mide su \textbf{error relativo porcentual}
              $\varepsilon(t) = 100\,(E(t) - E_0)/E_0$, y como escalar
              $\varepsilon_{\max}(\dt) = \max_t |\varepsilon(t)|$.
        \item \rotulo{Distribución del módulo de la velocidad}: $f(v)$, histograma
              de $|\vv|$ en 40 intervalos (área 1) sobre \textbf{una sola bolsa}
              con los $|\vv|$ de las $N$ partículas de las $M$ realizaciones; la
              \textbf{gran bolsa} del estacionario suma además todos los cuadros con
              $t \ge \SI{10}{\second}$ y se ajusta con \textbf{Maxwell-Boltzmann} en 2D,
              $f(v) = (m v / k_BT)\,e^{-m v^2 / 2 k_BT}$, con $k_BT$ como único
              parámetro: el mínimo del error cuadrático medio $E(k_BT)$ (Teórica 0).
              La rapidez media $\langle |\vv| \rangle$ de cada instante dice desde
              cuándo es estacionario.
        \item \rotulo{Sistema 1}: $r(t)$ contra la analítica y
              $\displaystyle \mathrm{ECM}(\dt) = \frac{1}{n}
              \sum_{j=1}^{n}\left[r_{\text{num}}(t_j) - r_{\text{an}}(t_j)\right]^2$
              sobre los $n$ pasos de integración.
        \item \rotulo{Promedios}: $\langle x \rangle = \frac{1}{M}\sum_{k=1}^{M} x_k$
              sobre $M$ realizaciones; la barra de error es el \textbf{desvío
              estándar} entre ellas.
    \end{itemize}
\end{frame}

\section{Resultados}
% =========================================================

\begin{frame}{Sistema 1: posición contra el tiempo}
    \figbulw[0.3em]{slide-13-posicion-oscilador-vs-t.png}{%
        \item Lo que se estima es $r(t)$: los cuatro esquemas sobre la analítica
              $r(t) = A\,e^{-\gamma t / 2m} \cos \omega t$, con el paso más largo
              del barrido.
        \item \textbf{Beeman}, \textbf{Verlet original} y \textbf{Velocity Verlet} caen
              sobre la analítica: se apartan como mucho $1$, $1$ y \SI{6}{\milli\meter}
              (sobre $A = \SI{1}{\meter}$).
        \item \alert{Euler p-c pierde amplitud}: se aparta hasta \SI{15}{\centi\meter}
              y a \SI{1}{\second} ya está \SI{13}{\centi\meter} por debajo. Con pasos más
              cortos se acerca: lo mide el ECM.
        \cond{$\dt = \SI{5e-3}{\second}$; se muestran los primeros \SI{2}{\second} de $t_f = \SI{5}{\second}$}
    }
\end{frame}

\begin{frame}{Sistema 1: error cuadrático medio contra el paso}
    \figbulw[0.2em]{slide-14-ECM-vs-dt.png}{%
        \item \textbf{Beeman} y \textbf{Verlet original}: ECM $\propto \dt^{4}$ (pendiente
              ajustada $4{,}0$), o sea error $\propto \dt^{2}$: se superponen.
        \item \textbf{Velocity Verlet} y \textbf{Euler p-c}: ECM $\propto \dt^{2}$
              ($2{,}0$ y $1{,}9$), un orden menos: lo paga la $\vv$ con la que
              cada uno evalúa $\va(t{+}\dt)$.
        \item \alert{Beeman es el mejor}: menor ECM en 8 de los 9 pasos. En
              $\dt = \SI{1e-5}{\second}$ Verlet ya mide el \textbf{piso de redondeo}
              y queda 50 veces por encima.
        \cond{$m = \SI{70}{\kilogram}$, $k = \SI{1e4}{\newton\per\meter}$, $\gamma = \SI{100}{\kilogram\per\second}$, $A = \SI{1}{\meter}$, $t_f = \SI{5}{\second}$}
    }
\end{frame}

\begin{frame}{Conservación de la energía}
    \figbulw[0.25em]{slide-15-energia-vs-t.png}{%
        \item Conservativo: $E$ tiene que quedarse en
              $E_0 = N\,\tfrac{1}{2} m v_0^2 = \SI{3.75}{\joule}$; lo que se aparta
              es \textbf{error del esquema}.
        \item $\SI{2e-4}{\second}$ \textbf{baja} hasta \SI{3.73}{\joule};

\section{Conclusiones}
% =========================================================

\begin{frame}{Conclusiones}
    \vspace{0.1em}
    \begin{itemize}\setlength{\itemsep}{0.75em}
        \item Beeman y Verlet original: $\mathrm{ECM} \propto \dt^{4}$; Velocity Verlet
              y Euler p-c: $\propto \dt^{2}$. Beeman es el mejor para el oscilador.
        \item Por conservación de la energía, $\dt = \SI{5e-5}{\second}$: el error
              de $E$ queda en \SI{0.04}{\percent} a $N = 300$ y \SI{0.03}{\percent}
              a $N = 700$.
        \item El costo del paso temporal crece casi lineal con $N$ y el de los eventos
              mucho más rápido: se cruzan cerca de $N = 470$ y a $N = 600$ los eventos
              cuestan $2{,}9$ veces más.
        \item Hay un $x_o$ \alert{óptimo} en $0{,}2$--$\SI{0.3}{\meter}$:
              \SI{14.8}{\second} contra \SI{31.2}{\second} en la pared, y es el mismo
              ($0{,}237 \pm \SI{0.011}{\meter}$) de $N = 20$ a $300$.
        \item $f(v)$ llega a Maxwell-Boltzmann en medio segundo, con
              $k_BT = \SI{12.4}{\milli\joule}$: el $m v_0^2/2$ del arranque, y
              $\langle |\vv| \rangle$ medido a \SI{0.2}{\percent} del que predice.
        \item Con los obstáculos en el centro hay una densidad óptima,
              $\rho = \SI{86}{\particulas\per\meter\squared}$ ($N = 70$):
              $\mtnov = \SI{21.8}{\second}$, contra \SI{66}{\second} a
              $\rho = 6$ y \SI{54.3}{\second} a $\rho = 367$.
    \end{itemize}
\end{frame}
