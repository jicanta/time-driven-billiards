\section{Introducción}
% =========================================================

\begin{frame}{Sistema}
    \begin{columns}[c,onlytextwidth]
        \begin{column}{0.50\textwidth}
            \large
            \begin{itemize}\setlength{\itemsep}{1.0em}
                \item Billar \textbf{circular} de radio $R$ con pared fija y dos
                      \textbf{obstáculos} fijos de radio $r$ en $(\pm x_o, 0)$.
                \item $N$ partículas de radio $r$ y masa $m$, rapidez inicial $v_0$
                      y dirección al azar.
                \item Interactúan \textbf{sólo por contacto}, sin fricción ni
                      disipación.
                \item Nacen \textcolor{fresca}{\textbf{frescas}} y pasan a
                      \textcolor{usada}{\textbf{usadas}} al tocar \textbf{por primera
                      vez} un obstáculo; siguen moviéndose igual, $N$ no cambia.
            \end{itemize}
        \end{column}\hfill
        \begin{column}{0.47\textwidth}
            \centering
            \input{figuras/sistema.tikz}\\[0.4em]
            {\footnotesize\color{gris}
             \textcolor{fresca}{$\bullet$} fresca \quad
             \textcolor{usada}{$\bullet$} usada \quad
             \textcolor{obstaculo}{$\bullet$} obstáculo}
        \end{column}
    \end{columns}
\end{frame}


\begin{frame}{Modelo: contacto elástico y pared como partícula imagen}
    \begin{columns}[c,onlytextwidth]
        \begin{column}{0.58\textwidth}
            \small
            \setlength{\abovedisplayskip}{4pt}\setlength{\belowdisplayskip}{4pt}
            Newton para cada partícula, $m\,\ddot{\vr}_i = \vF_i$, con un
            \textbf{resorte lineal} normal, activo sólo mientras hay solapamiento:
            \[
                \vF_i = \sum_j -k\,\xi_{ij}\,\ehat_{ij}, \qquad
                \xi_{ij} = r_i + r_j - |\vr_j - \vr_i| > 0, \qquad
                \ehat_{ij} = \frac{\vr_j - \vr_i}{|\vr_j - \vr_i|}
            \]
            Los $j$ son las otras partículas, los dos obstáculos (fijos) y la
            partícula imagen de la pared.
            \vspace{0.5em}

            \rotulo{Pared.} Hay contacto si $|\vr_i| > R - r$; la normal es radial,
            $\nhat = \vr_i/|\vr_i|$, y el choque es contra una partícula virtual fija
            de radio $r$ que toca la pared en $C = R\,\nhat$:
            \[
                \vr_{\text{img}} = (R + r)\,\nhat, \qquad
                \xi_{iw} = |\vr_i| + r - R .
            \]
            La fuerza queda puramente normal; $\vr_{\text{img}}$ se recalcula en cada
            paso (la partícula se desplaza tangencialmente durante el choque).
        \end{column}\hfill
        \begin{column}{0.40\textwidth}
            \centering
            \input{figuras/modelo-pared-imagen.tikz}\\[0.3em]
            {\footnotesize\color{gris} esquema no a escala}
        \end{column}
    \end{columns}
\end{frame}

\section{Implementación}
% =========================================================

\begin{frame}{Integración de las ecuaciones de movimiento}
    \vspace{-0.3em}
    \begin{columns}[t,onlytextwidth]
        \begin{column}{0.44\textwidth}
            \rotulo{Billar (sistema 2): Verlet original}
            \vspace{-0.4em}
            \[\begin{aligned}
                \vr(t{+}\dt) &= 2\vr(t) - \vr(t{-}\dt) + \tfrac{\dt^2}{m}\vF(t)\\[0.2em]
                \vv(t) &= \frac{\vr(t{+}\dt) - \vr(t{-}\dt)}{2\dt}
            \end{aligned}\]
            \vspace{-0.6em}
            {\small
            \begin{itemize}\setlength{\itemsep}{0.25em}
                \item Paso fijo $\dt$; el estado se guarda cada $\dtd = n\,\dt$,
                      con $n$ entero.
                \item Arranque: $\vr(-\dt)$ con Euler en $-\dt$. $\vF$ depende sólo
                      de las posiciones: paso explícito.
                \item Tres posiciones por partícula: $\vr(t)$ y $\vv(t)$ del
                      \textbf{mismo instante}.
            \end{itemize}}
        \end{column}\hfill
        \begin{column}{0.53\textwidth}
            \rotulo{Oscilador (sistema 1): además}
            \vspace{-0.5em}
            {\scriptsize
            \[\begin{aligned}
                &\textbf{Velocity Verlet}\\
                &\vr(t{+}\dt) = \vr(t) + \dt\,\vv(t) + \tfrac{\dt^2}{2}\va(t), \quad
                 \vv(t{+}\tfrac{\dt}{2}) = \vv(t) + \tfrac{\dt}{2}\va(t)\\
                &\vv(t{+}\dt) = \vv(t{+}\tfrac{\dt}{2}) + \tfrac{\dt}{2}\va(t{+}\dt)\\[0.15em]
                &\textbf{Beeman (predictor-corrector)}\\
                &\vr(t{+}\dt) = \vr(t) + \vv(t)\dt + \tfrac{2}{3}\va(t)\dt^2 - \tfrac{1}{6}\va(t{-}\dt)\dt^2\\
                &\vv^{\,p}(t{+}\dt) = \vv(t) + \tfrac{3}{2}\va(t)\dt - \tfrac{1}{2}\va(t{-}\dt)\dt\\
                &\vv(t{+}\dt) = \vv(t) + \tfrac{1}{3}\va(t{+}\dt)\dt + \tfrac{5}{6}\va(t)\dt - \tfrac{1}{6}\va(t{-}\dt)\dt\\[0.15em]
                &\textbf{Euler predictor-corrector}\\
                &\vv^{\,p} = \vv(t) + \va(t)\dt, \quad \vr^{\,p} = \vr(t) + \vv(t)\dt\\
                &\vv(t{+}\dt) = \vv(t) + \va(t{+}\dt)\dt, \quad \vr(t{+}\dt) = \vr(t) + \vv(t{+}\dt)\dt
            \end{aligned}\]}
        \end{column}
    \end{columns}
    \vspace{-0.1em}
    {\footnotesize
    En el oscilador $\vF = -k\vr - \gamma\vv$ \textbf{depende de la velocidad}:
    $\va(t{+}\dt)$ necesita $\vv(t{+}\dt)$, y hay que elegir con qué $\vv$ evaluarla.
    Verlet original la despeja; Beeman, la predicha, $O(\dt^2)$; \textbf{Velocity
    Verlet}, la de medio paso, $O(\dt)$, y \alert{pierde un orden}; Euler p-c, la
    de Euler.}
\end{frame}

\begin{frame}{Observables}
    \vspace{-0.2em}
    \small
    \begin{itemize}\setlength{\itemsep}{0.7em}
        \item \rotulo{Conversión}: $\Fu(t) = \Nu(t)/N$, con $\Nu$ las usadas
              acumuladas; \; $\tnov = \min\{\,t : \Fu(t) \ge 0{,}9\,\}$ \; y \;
              $\tcien = \min\{\,t : \Fu(t) = 1\,\}$, en el instante exacto de cada
              conversión.
        \item \rotulo{Energía total}, sobre los estados guardados cada $\dtd$:
              $\displaystyle E(t) = \sum_{i} \tfrac{1}{2} m v_i^2
               + \sum_{\text{contactos}} \tfrac{1}{2} k\,\xi^2$; \;
              deriva relativa máxima $D(\dt) = \max_t |E(t) - E_0|/E_0$.
        \item \rotulo{Distribución del módulo de la velocidad}: $f(v)$, histograma de
              $|\vv|$ en 40 intervalos (área 1) con las $N$ partículas de las $M$
              realizaciones en un instante; en el estacionario, además todos los
              cuadros con $t \ge \SI{10}{\second}$.
        \item \rotulo{Sistema 1}: $\displaystyle \mathrm{ECM}(\dt) = \frac{1}{n}
              \sum_{j=1}^{n}\left[r_{\text{num}}(t_j) - r_{\text{an}}(t_j)\right]^2$
              sobre los $n$ pasos de integración.
        \item \rotulo{Promedios}: $\langle x \rangle = \frac{1}{M}\sum_{k=1}^{M} x_k$
              sobre $M$ realizaciones; la barra de error es el \textbf{desvío
              estándar} entre ellas.
    \end{itemize}
\end{frame}

\section{Resultados}
% =========================================================

\begin{frame}{Sistema 1: error cuadrático medio contra el paso}
    \figbulw[0.2em]{slide-13-ECM-vs-dt.png}{%
        \item \textbf{Beeman} y \textbf{Verlet original}: ECM $\propto \dt^{4}$, o sea
              error $\propto \dt^{2}$. En esta escala se superponen.
        \item \textbf{Velocity Verlet} y \textbf{Euler p-c}: ECM $\propto \dt^{2}$, un
              orden menos. A $\dt = \SI{1e-4}{\second}$, ocho órdenes hasta Beeman.
        \item \alert{Beeman es el mejor}: menor ECM en 8 de los 9 pasos. Lo que lo
              separa de Verlet es la constante del error, un \SI{15}{\percent}.
        \cond{$m = \SI{70}{\kilogram}$, $k = \SI{1e4}{\newton\per\meter}$, $\gamma = \SI{100}{\kilogram\per\second}$, $A = \SI{1}{\meter}$, $t_f = \SI{5}{\second}$}
    }
\end{frame}

\begin{frame}{Energía total contra el tiempo}
    \figbulw[0.35em]{slide-14-energia-vs-t.png}{%
        \item $E_0 = N\,\tfrac{1}{2} m v_0^2 = \SI{3.75}{\joule}$: el sistema es
              conservativo y se mide cuánto se aparta.
        \item $\SI{5e-4}{\second}$ deriva un \SI{2.6}{\percent} y
              $\SI{2e-4}{\second}$ un \SI{0.6}{\percent} (hacia abajo); de
              $\SI{1e-4}{\second}$ en adelante no se distinguen de $E_0$.
        \item Fuera de escala, $\dt = \SI{1e-3}{\second}$ explota a \SI{45}{\joule}:
              el choque dura $\tau = \pi\sqrt{\mu/k} = \SI{3.5}{\milli\second}$ y
              entran sólo tres pasos.
        \cond{$N = 300$, sin obstáculos, $t_f = \SI{5}{\second}$, $\dtd = \SI{0.01}{\second}$}
    }
\end{frame}

\begin{frame}{Elección del paso: deriva relativa máxima}
    \figbul{slide-15-deriva-vs-dt.png}{%
        \item Observable escalar: la deriva relativa máxima $D(\dt)$. Sobre los
              pasos estables ajusta $D \propto \dt^{2,0}$: el segundo orden de Verlet.
        \item Criterio: el $\dt$ \textbf{más largo} con $D \leq 10^{-3}$, el de menor
              costo con el error aceptado.
        \item \alert{$\dt = \SI{5e-5}{\second}$}: $D = 3{,}8\cdot10^{-4}$, y
              $2{,}6\cdot10^{-4}$ en el caso más exigente, $N = 700$ con obstáculos.
        \cond{$N = 300$, sin obstáculos, $t_f = \SI{5}{\second}$}
    }
\end{frame}

\begin{frame}{Obstáculos en el centro}

\section{Conclusiones}
% =========================================================

\begin{frame}{Conclusiones}
    \vspace{0.1em}
    \begin{itemize}\setlength{\itemsep}{0.75em}
        \item Beeman y Verlet original: $\mathrm{ECM} \propto \dt^{4}$; Velocity Verlet
              y Euler p-c: $\propto \dt^{2}$. Beeman es el mejor para el oscilador.
        \item Por conservación de la energía, $\dt = \SI{5e-5}{\second}$: deriva
              $3{,}8\cdot10^{-4}$ a $N = 300$ y $2{,}6\cdot10^{-4}$ a $N = 700$.
        \item El costo del paso temporal crece casi lineal con $N$ y el de los eventos
              mucho más rápido: se cruzan cerca de $N = 470$ y a $N = 600$ los eventos
              cuestan $2{,}9$ veces más.
        \item Hay un $x_o$ \alert{óptimo} en $0{,}2$--$\SI{0.3}{\meter}$:
              \SI{14.8}{\second} contra \SI{31.2}{\second} en la pared; es el mismo
              de $N = 20$ a $300$.
        \item $f(v)$ llega a Maxwell-Boltzmann en medio segundo, con
              $k_BT = \SI{12.4}{\milli\joule}$: el $m v_0^2/2$ del arranque.
        \item Con los obstáculos en el centro hay una densidad óptima, $N = 30$--$50$;
              a $N = 700$ se convierte el \SI{1}{\percent} en \SI{30}{\second}.
    \end{itemize}
\end{frame}
